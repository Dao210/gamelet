## 🌟 一、核心理念

### 1.1 创意

**全球草原花园 (Global Prairie Garden)** ✨
- **空间形态**: 水平延展，共享空间
- **共生性**: 和谐共生，相互陪伴
- **视觉隐喻**: 一片广袤的草原，每朵花都有自己的位置
- **情感基调**: 温暖、治愈、归属感、陪伴

### 1.2 核心价值主张

> **"在这片草原上，每一株植物都沐浴着同样温暖的阳光，感受着同样轻柔的微风。你的创作不是孤独的，它与来自世界各地的植物们一起，构成了这片生机勃勃的花园。"**

**三大核心价值**:
1. **创作自由** - 用简单的工具，画出独特的植物
2. **情感连接** - 每一次浇水都是一份温暖的陪伴
3. **共生美学** - 在广袤的草原上，见证生命的多样性与和谐

#### 微风系统
```typescript
interface WindSystem {
  direction: number // 0-360度
  strength: number // 0-1
  gustInterval: number // 阵风间隔(ms)
}

/**
 * 风力驱动草和植物的摆动
 * 影响范围:
 * - 草地摆动 (sway animation)
 * - 植物摇曳 (plant sway)
 * - 云朵移动 (cloud drift)
 */
function applyWind(element: HTMLElement, wind: WindSystem) {
  const baseRotation = wind.direction
  const amplitude = wind.strength * 15 // 最大15度摆动
  const duration = 2000 - (wind.strength * 1000) // 风越大，摆动越快

  element.style.animation = `
    sway ${duration}ms ease-in-out infinite
  `
  element.style.setProperty('--rotation', `${baseRotation}deg`)
  element.style.setProperty('--amplitude', `${amplitude}deg`)
}
```

## 🌱 四、生长机制设计

### 4.1 双引擎生长系统

```typescript
interface PlantGrowth {
  socialGrowth: {
    xp: number // 社交经验值
    waterCount: number // 被浇水次数
    uniqueWaterers: string[] // 独立浇水用户
  }
  timeGrowth: {
    createdAt: Date
    daysSinceCreation: number
    ageBonus: number // 时间加成 (每天+5 XP)
  }
}

/**
 * 双引擎计算公式:
 * totalXP = socialXP + timeXP
 * socialXP = waterCount * 10
 * timeXP = daysSinceCreation * 5
 *
 * 设计理念:
 * - 社交引擎: 鼓励互动和传播
 * - 时间引擎: 保护冷门但用心的作品
 */
function calculateTotalXP(plant: PlantGrowth): number {
  const socialXP = plant.socialGrowth.waterCount * 10
  const timeXP = plant.timeGrowth.daysSinceCreation * 5
  return socialXP + timeXP
}
```

### 5.2 昼夜循环系统

```typescript
interface TimeOfDay {
  hour: number // 0-23

/**

  },
  {
    hour: 17,
    phase: 'afternoon',
    skyGradient: ['#FFD93D', '#FFA500'],
    lightIntensity: 0.8,
    shadowAngle: 120
  },
  {
    hour: 19,
    phase: 'dusk',
    skyGradient: ['#FF6B6B', '#C44569', '#5F27CD'],
    lightIntensity: 0.3,
    shadowAngle: 150
  },
  {
    hour: 21,
    phase: 'night',
    skyGradient: ['#0F2027', '#203A43', '#2C5364'],
    lightIntensity: 0.1,
    shadowAngle: 180
  }
]

/**
 * 平滑过渡实现 (每小时渐变15分钟)
 */
function updateTimeOfDay() {
  const now = new Date()
  const currentHour = now.getHours()
  const currentMinute = now.getMinutes()

  const currentPhase = timePhases.find(p => p.hour === currentHour)
  const nextPhase = timePhases.find(p => p.hour === (currentHour + 1) % 24)

  if (!currentPhase || !nextPhase) return

  // 在每小时的最后15分钟开始过渡
  if (currentMinute >= 45) {
    const progress = (currentMinute - 45) / 15 // 0-1
    const transitionedGradient = interpolateGradient(
      currentPhase.skyGradient,
      nextPhase.skyGradient,
      progress
    )
    applySkyGradient(transitionedGradient)
  } else {
    applySkyGradient(currentPhase.skyGradient)
  }
}

// 每分钟更新一次
setInterval(updateTimeOfDay, 60000)
```

### 6.3 关键算法实现

#### Poisson Disk Sampling (植物布局)
```typescript
/**
 * Bridson's Algorithm 实现
 * 时间复杂度: O(n)
 * 空间复杂度: O(n)
 * 参考: https://www.cs.ubc.ca/~rbridson/docs/bridson-siggraph07-poissondisk.pdf
 */

interface Point {
  x: number
  y: number
}

class PoissonDiskSampling {
  private readonly width: number
  private readonly height: number
  private readonly minDistance: number
  private readonly maxAttempts: number
  private readonly cellSize: number
  private readonly gridWidth: number
  private readonly gridHeight: number
  private readonly grid: (Point | null)[][]

  constructor(
    width: number,
    height: number,
    minDistance: number,
    maxAttempts: number = 30
  ) {
    this.width = width
    this.height = height
    this.minDistance = minDistance
    this.maxAttempts = maxAttempts

    // Grid cell size = minDistance / sqrt(2)
    this.cellSize = minDistance / Math.sqrt(2)
    this.gridWidth = Math.ceil(width / this.cellSize)
    this.gridHeight = Math.ceil(height / this.cellSize)

    // Initialize grid
    this.grid = Array(this.gridHeight)
      .fill(null)
      .map(() => Array(this.gridWidth).fill(null))
  }

  generate(): Point[] {
    const points: Point[] = []
    const activeList: Point[] = []

    // Step 1: Select initial sample
    const initial = {
      x: Math.random() * this.width,
      y: Math.random() * this.height
    }
    points.push(initial)
    activeList.push(initial)
    this.insertIntoGrid(initial)

    // Step 2: While active list is not empty
    while (activeList.length > 0) {
      const randomIndex = Math.floor(Math.random() * activeList.length)
      const point = activeList[randomIndex]
      let found = false

      // Try to generate maxAttempts new points
      for (let i = 0; i < this.maxAttempts; i++) {
        const newPoint = this.generateRandomPointAround(point)

        if (
          this.isValid(newPoint) &&
          this.isDistanceValid(newPoint)
        ) {
          points.push(newPoint)
          activeList.push(newPoint)
          this.insertIntoGrid(newPoint)
          found = true
          break
        }
      }

      // Remove from active list if no valid point found
      if (!found) {
        activeList.splice(randomIndex, 1)
      }
    }

    return points
  }

  private generateRandomPointAround(point: Point): Point {
    // Generate point in annulus between minDistance and 2*minDistance
    const angle = Math.random() * 2 * Math.PI
    const radius =
      this.minDistance * (1 + Math.random()) // [minDistance, 2*minDistance]

    return {
      x: point.x + radius * Math.cos(angle),
      y: point.y + radius * Math.sin(angle)
    }
  }

  private isValid(point: Point): boolean {
    return (
      point.x >= 0 &&
      point.x < this.width &&
      point.y >= 0 &&
      point.y < this.height
    )
  }

  private isDistanceValid(point: Point): boolean {
    const gridX = Math.floor(point.x / this.cellSize)
    const gridY = Math.floor(point.y / this.cellSize)

    // Check neighboring cells (5x5 grid around point)
    for (let dy = -2; dy <= 2; dy++) {
      for (let dx = -2; dx <= 2; dx++) {
        const nx = gridX + dx
        const ny = gridY + dy

        if (
          nx >= 0 &&
          nx < this.gridWidth &&
          ny >= 0 &&
          ny < this.gridHeight
        ) {
          const neighbor = this.grid[ny][nx]
          if (neighbor) {
            const dist = this.distance(point, neighbor)
            if (dist < this.minDistance) {
              return false
            }
          }
        }
      }
    }

    return true
  }

  private insertIntoGrid(point: Point): void {
    const gridX = Math.floor(point.x / this.cellSize)
    const gridY = Math.floor(point.y / this.cellSize)
    this.grid[gridY][gridX] = point
  }

  private distance(a: Point, b: Point): number {
    const dx = a.x - b.x
    const dy = a.y - b.y
    return Math.sqrt(dx * dx + dy * dy)
  }
}

// Usage
const sampler = new PoissonDiskSampling(1920, 10000, 150, 30)
const positions = sampler.generate()
```

```typescript
/**
 * 使用 @tanstack/react-virtual 实现虚拟滚动
 * 只渲染视口内+缓冲区的植物，大幅提升性能
 */

import { useVirtualizer } from '@tanstack/react-virtual'

function PrairieGarden({ plants }: { plants: Plant[] }) {
  const parentRef = useRef<HTMLDivElement>(null)

  // 虚拟滚动配置
  const virtualizer = useVirtualizer({
    count: plants.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 200, // 每行估计高度
    overscan: 3, // 缓冲区：视口外3行
    horizontal: false,
    // 自定义渲染范围计算
    measureElement:
      typeof window !== 'undefined' &&
      navigator.userAgent.indexOf('Firefox') === -1
        ? (element) => element?.getBoundingClientRect().height
        : undefined
  })

  const items = virtualizer.getVirtualItems()

  return (
    <div
      ref={parentRef}
      style={{
        height: '100vh',
        overflow: 'auto'
      }}
    >
      <div
        style={{
          height: `${virtualizer.getTotalSize()}px`,
          width: '100%',
          position: 'relative'
        }}
      >
        {items.map((virtualItem) => {
          const plant = plants[virtualItem.index]
          return (
            <div
              key={virtualItem.key}
              data-index={virtualItem.index}
              ref={virtualizer.measureElement}
              style={{
                position: 'absolute',
                top: 0,
                left: 0,
                width: '100%',
                transform: `translateY(${virtualItem.start}px)`
              }}
            >
              <PlantCard plant={plant} />
            </div>
          )
        })}
      </div>
    </div>
  )
}
```

## 📚 十五、附录

### A. 技术名词解释

- **Poisson Disk Sampling**: 一种空间点分布算法，确保点之间保持最小距离且分布均匀
- **Parallax Scrolling**: 视差滚动，不同层以不同速度移动，营造景深感
- **Virtual Scrolling**: 虚拟滚动，只渲染视口内的元素，提升性能
- **OffscreenCanvas**: 浏览器API，允许Canvas在Web Worker中渲染
- **Perlin Noise**: 一种自然纹理生成算法，常用于地形、云朵等
